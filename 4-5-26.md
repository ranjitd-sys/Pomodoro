function readCookie(name: string) {
  if (typeof document === "undefined") {
    return undefined;
  }
  const target = `${name}=`;
  const entry = document.cookie
    .split("; ")
    .find((part) => part.startsWith(target));
  return entry ? decodeURIComponent(entry.slice(target.length)) : undefined;
}

function readPathname(): string {
  return typeof window !== "undefined" ? window.location.pathname : "";
}

const SHIPPED_LESSON_SLUGS = new Set([
  "execute",
  "discard-poll",
  "idempotency",
  "memory-vs-opfs",
  "interrupt",
  "timeline",
]);

export function deriveRoute(actorId: string): string | undefined {
  if (actorId === lessonSlotId) return "workflow";
  if (actorId === workflowInvokeId && readPathname().startsWith("/workflow"))
    return "workflow";
  return undefined;
}

export function deriveLesson(_actorId: string): string | undefined {
  const path = readPathname();
  if (!path.startsWith("/workflow/")) return undefined;
  const slug = path.slice("/workflow/".length).split("/")[0];
  return slug && SHIPPED_LESSON_SLUGS.has(slug) ? slug : undefined;
}

function actorRefId(actorRef: InspectionEvent["actorRef"]): string {
  const candidate = actorRef as unknown as { id?: string };
  return candidate.id ?? "";
}

export function shouldEmit(event: InspectionEvent): boolean {
  const route = deriveRoute(actorRefId(event.actorRef));
  if (route !== "workflow") return false;

  switch (event.type) {
    case "@xstate.event":
      if (globalThis.String(event.event.type).startsWith("xstate."))
        return false;
      return true;
    case "@xstate.snapshot":
      return false;
    case "@xstate.actor":
      return true;
    default:
      return false;
  }
}
