**on vault**

```
apiVersion: barmancloud.cnpg.io/v1
kind: ObjectStore
metadata:
  name: deepecom-backups
  namespace: default
spec:
  configuration:
    destinationPath: "s3://deepecom-cnpg-backups/"
    endpointURL: "https://blr1.vultrobjects.com"
    s3Credentials:
      accessKeyId:
        name: vultr-s3-creds
        key: ACCESS_KEY_ID
      secretAccessKey:
        name: vultr-s3-creds
        key: SECRET_ACCESS_KEY
    wal:
      compression: gzip
    data:
      compression: gzip
  retentionPolicy: "1d"
```

```
apiVersion: barmancloud.cnpg.io/v1
kind: ObjectStore
metadata:
  name: deepecom-backups

        name: azure-creds
```
