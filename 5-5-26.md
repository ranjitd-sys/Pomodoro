**Updated Implementation**
``` 

import { Context, Effect, Layer } from "effect"
import { StreamChat } from "stream-chat";

class StreamService extends Context.Service<StreamService, {
  readonly prefix: string;
}>()("app/Config", {
  make: Effect.gen(function* () {
     const apiKey = process.env.API_KEY!;
      const apiSecret = process.env.API_SECRET!;

      yield* Effect.log("Connecting Stream client...");

      const client = StreamChat.getInstance(apiKey, apiSecret);

    return { prefix: "MyApp" };
  }),
}) {
  static readonly layer = Layer.effect(StreamService, StreamService.make);
}
class Logger extends Context.Service<Logger>()("Logger", {
  make: Effect.gen(function*() {
    const streamService = yield* StreamService;
    return { log: (msg: string) => Effect.log(`[${streamService.prefix}] ${msg}`) }
  })
}) {
  // Build the layer yourself from the make effect
  static readonly layer = Layer.effect(this, this.make).pipe(
    Layer.provide(StreamService.layer)
  )
}
```
