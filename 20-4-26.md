*main workflow*
```
import { Effect, Layer } from "effect";
import { WorkflowProxy } from "effect/unstable/workflow";
import {
  ClusterWorkflowEngine, Runners, Sharding,
  SqlRunnerStorage, Snowflake, SqlMessageStorage,
  RunnerHealth, ShardingConfig,
} from "effect/unstable/cluster";
import { RpcClientProtocolLive } from "./layers/rpc";
import { SqlLayer } from "./layers/sql";
import { RpcSerialization } from "effect/unstable/rpc";
import { incrementWorkflow } from "./flipkart/contract";
import { incrementWorkflowLayer } from "./flipkart/handler";


const ShardingConfigLive = ShardingConfig.layer({
  shardsPerGroup: 50,
  entityMailboxCapacity: 10,
  entityTerminationTimeout: 30_000,
  entityMessagePollInterval: 1000,
  sendRetryInterval: 100,
});

export const workflows = [incrementWorkflow] as const;

export class WorkflowRpc extends WorkflowProxy.toRpcGroup(workflows) {}

export const WorkflowEngine = ClusterWorkflowEngine.layer.pipe(
  Layer.provideMerge(Sharding.layer),
  
  Layer.provideMerge(
    Runners.layerRpc.pipe(Layer.provide(RpcClientProtocolLive)),
  ),
  Layer.provideMerge(
    SqlMessageStorage.layer.pipe(
      Layer.provide(Snowflake.layerGenerator),
      Layer.provide(SqlLayer),
    ),
  ),
  Layer.provide(
    SqlRunnerStorage.layer.pipe(
      Layer.provide(Snowflake.layerGenerator),
      Layer.provide(SqlLayer),
    ),
  ),
  Layer.provide(ShardingConfigLive),
  Layer.provide(RunnerHealth.layerNoop),
);

export const FullLayer = Layer.mergeAll(
  incrementWorkflowLayer,
  SqlLayer,
  WorkflowEngine,
);
```
