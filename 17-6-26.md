1. [x] **Added index.html to see the event on web**
```html
<!DOCTYPE html>
<html>
<body>
  <input id="sid" placeholder="stream id" />
  <button onclick="connect()">Connect</button>
  <pre id="out"></pre>
  <script>
    let es;

    function connect() {
      if (es) es.close();

      const sid = document.getElementById("sid").value;
      const out = document.getElementById("out");
      out.textContent = "";

      es = new EventSource(`http://20.87.39.196/api/sync/${sid}?offset=-1`);

      es.onmessage = (e) => {
        out.textContent += e.data + "\n";
      };

      es.addEventListener("done", () => {
        out.textContent += "\n--- stream done ---";
        es.close();  // <-- closes here when server signals done
      });

      es.onerror = () => {
        out.textContent += "\n--- error ---";
        es.close();  // <-- also close on error so it doesn't reconnect
      };
    }
  </script>
</body>
</html>
```
