    streamRegistryAtom(appAtomRegistry, workflowProbeAtom),
  );
  const machine = setup({
    types: {
      input: {} as { registry: AtomRegistryInstance },
      events: {} as {
        type: "send";
        source: string;
        destination: string;
        amount: string;
      },
    },
    actors: {

    context: ({ input }) => ({
      registry: input.registry,
      source: "",
      destination: "",
      amount: 0,
      probe: input.registry.get(workflowProbeAtom),
    }),
    initial: "ready",
    states: {
      ready: {
        invoke: {
          src: "PROBE",
          onSnapshot: {
            actions: assign({
              probe: ({ context, event }) =>
                event["snapshot"]["context"] ?? context["probe"],
            }),
          },
        },

        on: {
          send: {
            target: "transferring",
            actions: assign({
              source: ({ event }) => event.source,
              destination: ({ event }) => event.destination,
              amount: ({ event }) => Number(event.amount),
            }),
          },
        },
      },
      transferring: {
        invoke: {
          src: userActor,
          input: ({ context }) => ({
            source: context["source"],
            destination: context["destination"],
            amount: context["amount"],
          }),
          onDone: {
            target: "validate",
            actions: assign({ result: ({ event }) => event.output }),
          },
          onError: {
            target: "error",
            actions: assign({ error: ({ event }) => event.error }),
          },
        },
      },
      validate: {
        invoke: {
          src: validateActor,
          input: ({ context }) => ({
            source: context["source"],
            destination: context["destination"],
            amount: context["amount"],

            target: "debit",

          },
          onError: {
            target: "error",
            actions: assign({ error: ({ event }) => event.error }),
          },

            destination: context["destination"],

          }),
          onDone: {
            target: "credit",

          },

    selectors: ({ ctx }) => ctx,
    actions: ({ send }) => ({
      send: (payload: {
        source: string;
        destination: string;
        amount: string;

    View: (props) => {
      const currentState = props.selectors["state"] ?? "ready";
      const [source, setSource] = useState("");
      const [destination, setDestination] = useState("");
      const [amount, setAmount] = useState("");

      const STEPS = [
        { label: "Fetch balance", state: "transferring" },
        { label: "Validate balance", state: "validate" },
        { label: "Subtract from source", state: "debit" },
        { label: "Add to destination", state: "credit" },
        { label: "Confirm updated balances", state: "success" },
      ];

      const activeIdx = STEPS.findIndex((s) => s.state === currentState);

      const getStepStatus = (i: number) => {
        if (currentState === "error") return i < activeIdx ? "done" : "idle";
        if (i < activeIdx) return "done";
        if (i === activeIdx) return "active";
        return "idle";
      };

      return (
        <div className="min-h-screen bg-[#0a0a0a] flex items-center justify-center text-white">
          <div className="w-full max-w-xl p-8">
            <h2 className="text-3xl font-semibold mb-8 tracking-tight">
              Money Transfer
            </h2>

            {/* Input Form */}
            <div className="mb-10 rounded-2xl border border-gray-800 bg-[#111] p-6 space-y-4">
              <p className="text-sm text-gray-400">Input</p>
              <input
                value={amount}
                onChange={(e) => setAmount(e.target.value)}
                placeholder="Amount (e.g. 100)"
                className="w-full bg-[#0a0a0a] border border-gray-700 rounded-lg px-4 py-2 text-sm outline-none focus:border-white transition"
              />
              <button
                onClick={() =>
                  props.actions.send({ source, destination, amount })
                }
                className="w-full mt-2 bg-white text-black rounded-lg py-2 text-sm font-medium hover:bg-gray-200 transition"
              >
                Run Workflow
              </button>
            </div>

            {/* Workflow Steps */}
            <div className="relative flex flex-col gap-6">
              <div className="absolute left-5 top-2 bottom-2 w-px bg-gradient-to-b from-transparent via-gray-700 to-transparent" />

              {STEPS.map((step, i) => {
                const status = getStepStatus(i);
                return (
                  <div key={i} className="flex items-center gap-5">
                    <div
                      className={`relative z-10 w-10 h-10 flex items-center justify-center rounded-full border transition-all duration-300
                    ${status === "active" ? "bg-white border-white text-black" : ""}
                    ${status === "done" ? "bg-green-500 border-green-500 text-black" : ""}
                    ${status === "idle" ? "bg-[#111] border-gray-700 text-gray-400" : ""}
                  `}
                    >
                      {status === "done" ? (
                        <span className="text-sm">✓</span>
                      ) : (
                        <span className="text-sm">{i + 1}</span>
                      )}
                    </div>

                    <p
                      className={`text-sm transition-colors duration-300
                    ${status === "active" ? "text-white font-medium" : ""}
                    ${status === "done" ? "text-green-400" : ""}
                    ${status === "idle" ? "text-gray-500" : ""}
                  `}
                    >
                      {step.label}
                    </p>
                  </div>
                );
              })}
            </div>

            {/* Output */}
            <div className="mt-10 rounded-2xl border border-gray-800 bg-[#111] p-5">
              <p className="text-sm text-gray-400 mb-2">Output</p>
              {currentState === "success" && (
                <p className="text-green-400 font-medium">
                  Success: Transfer Completed
                </p>
              )}
              {currentState === "error" && (
                <p className="text-red-400 font-medium">
                  Error: Transfer Failed
                </p>
              )}
              {!["success", "error"].includes(currentState) && (
                <p className="text-gray-500 text-sm">Waiting for transfer...</p>
              )}
            </div>
          </div>
        </div>
      );
    },
  });
});
