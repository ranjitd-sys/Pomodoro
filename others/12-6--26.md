export function parseReplyTexts(
  replyTexts: Map<string, any>,
): { event: WorkflowEvent; marketplace: string } | null {
  const values = [...replyTexts.values()] as [string, string][];

  const filePaths = [...replyTexts.values()].filter(
    (v): v is string => typeof v === "string" && v.includes(".pdf"),
  );
  const TERMINAL_PRIORITY = ["unauthorised", "uploaded", "failed"] as const;

  let lastEntry: [string, string] | undefined;

  for (const priority of TERMINAL_PRIORITY) {
    const found = values.find(
      ([stage]) => stage?.toLowerCase().trim() === priority,
    );
    if (found) { lastEntry = found; break; }
  }

  if (!lastEntry) lastEntry = values.at(-1);

  const [latestStage = "", latestMarketplace = ""] = lastEntry ?? [];
  const m = latestMarketplace.toLowerCase();

  if (latestStage.toLowerCase().trim() === "unauthorised") {
    const type =
      m === "flipkart" ? "FLIPKART_UNAUTHORIZED"
      : m === "meesho" ? "MEESHO_UNAUTHORIZED"
      : m === "firstcry" ? "FIRSTCRY_UNAUTHORIZED"
      : "UNAUTHORIZED";
    return { event: { type }, marketplace: latestMarketplace };
  }

  if (latestStage.toLowerCase().trim() === "uploaded") {
    const event: WorkflowEvent =
      m === "flipkart" ? { type: "FLIPKART_COMPLETED" }
      : m === "meesho" ? { type: "MEESHO_COMPLETED" }
      : { type: "FIRSTCRY_COMPLETED", filePaths };
    return { event, marketplace: latestMarketplace };
  }

  if (latestStage.toLowerCase().trim() === "failed") {
    const type =
      m === "flipkart" ? "FLIPKART_FAILED"
      : m === "meesho" ? "MEESHO_FAILED"
      : m === "firstcry" ? "FIRSTCRY_FAILED"
      : "UNAUTHORIZED";
    return { event: { type }, marketplace: latestMarketplace };
  }

  return null;
}

export function buildReplyTexts(
  replies: Array<{
    id: string;
    text?: string;
    attachments?: { asset_url?: string }[];
  }>,
): Map<string, any> {
  const replyTexts = new Map<string, any>();

  for (const reply of replies) {
    try {
      const parsed = JSON.parse(reply.text ?? "{}");
      replyTexts.set(reply.id, [parsed.stage, parsed.marketplace]);
    } catch {
      const url = reply.attachments?.[0]?.asset_url;
      if (url) replyTexts.set(reply.id, url);
      else replyTexts.set(reply.id, reply.text ?? "");
    }
  }

  return replyTexts;
}

export const marketplaceWatchActor = ({
  queue,
  onEvent,
  client,
}: {
  queue: Queue.Queue<QueueItem>;
  onEvent: (type: WorkflowEvent) => void;
  client: stream.StreamChat;
}) =>
  Actor.fromEffect(() =>
    Effect.gen(function* () {
      while (true) {
        yield* Effect.logInfo(`[Flipkart] Queue consumer`);
        yield* Effect.sleep("4 seconds");

        const size = yield* Queue.size(queue);
        yield* setGauge(flipkartQueueDepth, size);

        const items: QueueItem[] = [];
        for (let i = 0; i < size; i++) items.push(yield* Queue.take(queue));

        const now = Date.now();

        for (const item of items) {
          if (now - item.sentAt > TEN_MINUTES) {
            yield* Effect.log(`[Flipkart] Discarding expired ${item.userId}`);
            continue;
          }

          const startedAt = Date.now();

          const result = yield* Effect.tryPromise({
            try: async () => {
              const { message } = await client.getMessage(item.messageId);
              if (!message.reply_count) return null;

              const channel = client.channel(
                message.channel!.type,
                message.channel?.id,
              );
              const { messages: replies } = await channel.getReplies(
                item.messageId,
                { limit: 5 },
              );

              const sortedReplies = [...replies].sort(
                (a, b) =>
                  new Date(a.created_at!).getTime() -
                  new Date(b.created_at!).getTime(),
              );

              const replyTexts = buildReplyTexts(sortedReplies);
              return parseReplyTexts(replyTexts);
            },
            catch: (e) => new Error(`Failed to check message: ${e}`),
          }).pipe(
            Effect.retry({
              times: 3,
              schedule: Schedule.exponential("500 millis", 2),
            }),

            Effect.catchCause((e) =>
              Effect.gen(function* () {
                yield* Effect.logWarning(
                  `[Flipkart] message check failed after retries, re-queuing: ${e}`,
                );
                yield* inc(flipkartReplyFailed);
                yield* Queue.offer(queue, item);
                return null;
              }),
            ),
          );

          yield* inc(flipkartQueueProcessed);
          yield* inc(flipkartMessageProcessed);

          if (result) {
            const durationMs = Date.now() - startedAt;
            yield* rec(flipkartWorkflowDuration, durationMs);

            const platform = result.marketplace.toLowerCase() as
              | "flipkart"
              | "meesho"
              | "firstcry";

            if (result.event.type.includes("UNAUTHORIZED")) {
              yield* inc(flipkartUnauthorized);
              yield* trackUnauthorized(platform);
            }

            if (result.event.type.includes("FAILED")) {
              yield* trackFailed(platform);
            }

            if (result.event.type.includes("COMPLETED")) {
              yield* inc(pdfsParsed);
            }

            onEvent(result.event);
            yield* Effect.log(
              `[Flipkart] ${result.event.type} — ${item.userId}`,
            );
          } else {
            yield* Queue.offer(queue, item);
          }
        }
      }
    }).pipe(
      // If the whole actor crashes, restart it
      Effect.retry({
        schedule: Schedule.exponential("1 second", 2).pipe(
          Schedule.both(Schedule.spaced("30 seconds")),
        ),
      }),
      Effect.tapError((e) =>
        Effect.logError(`[Flipkart] actor crashed, restarting: ${e}`),
      ),
    ),
  );
