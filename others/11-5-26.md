2. [x] **Deploying on k8s and try to spawn on from Maruti**
```
import { Effect, Schedule } from "effect";
import { assign, createActor, createMachine } from "xstate";
import * as stream from "@de-workflow/getstream";
import * as Actor from "@de-workflow/xstate";
import { makeImportActor } from "../k8s/Import";
import * as k8s from "@kubernetes/client-node";

const NAMESPACE = "default";
const POD_NAME = "amazon-worker-import";
const IMAGE = "workflowdeepecom.azurecr.io/import:V0.2";

const Maruti = Effect.gen(function* () {
  const { client } = yield* stream.StreamService;

  let marutiActor: any;

  const watchActor = yield* Actor.fromEffect(() =>
    Effect.gen(function* () {
      yield* Effect.tryPromise({
        try: () =>
          client.connectUser(
            { id: "ranjit-dev" },
            client.createToken("ranjit-dev"),
          ),
        catch: (e) => new Error(`connectUser failed: ${e}`),
      });

      const channel = client.channel("messaging", "message");

      yield* Effect.tryPromise({
        try: () => channel.watch(),
        catch: (e) => new Error(`watch failed: ${e}`),
      });

      yield* Effect.log("Channel watching ✓");

      channel.on("message.new", (event) => {
        marutiActor.send({
          type: "MESSAGE_RECEIVED",
          message: event.message?.text,
        });
      });

      yield* Effect.never;
    }),
  );

  const processActor = yield* Actor.fromEffect(() =>
    Effect.gen(function* () {
      yield* Effect.log("processing...");
      yield* Effect.sleep("1 second");
    }),
  );

  const importActorLogic = yield* makeImportActor();
  
  const MarutiMachine = createMachine({
  id: "Maruti",
  type: "parallel",
  
  context: {
    amazonRef: null as any,
  },
  states: {

    watcher: {
      invoke: {
        src: watchActor,
        onError: {
          actions: ({ event }) => console.error("watchActor error:", event.error),
        },
      },
    },

  
    processor: {
      invoke: {
        src: processActor,
        onError: {
          actions: ({ event }) => console.error("processActor error:", event.error),
        },
      },
    },

    importFlow: {
      initial: "idle",
      states: {
        idle: {
          on: {
            MESSAGE_RECEIVED: {
              guard: ({ event }) => event.message === "Amazon",
              target: "importing",
              actions: () => console.log("Amazon message received, starting import..."),
            },
          },
        },
        importing: {
          invoke: {
            src: importActorLogic,
            onDone: {
              target: "idle",
              actions: () => console.log("Import done"),
            },
            onError: {
              target: "idle",
              actions: ({ event }) => console.error("Import error:", event.error),
            },
          },
        },
      },
    },
  },
});

  marutiActor = createActor(MarutiMachine, {
  
}).start();

}).pipe(Effect.provide(stream.StreamService.layer));
Effect.runPromise(Maruti);
```

3. [x] **Converting all the fetch to HttpClient**
```
/**
 * Import actor — refactored to use @effect/cluster K8sHttpClient
 *
 * What changed vs the original:
 *  - No manual k8sFetch / raw fetch / fs.readFileSync
 *  - Pod lifecycle (create → wait for ready → delete on scope exit) is
 *    handled entirely by makeCreatePod
 *  - Responses are schema-validated (no more `as any`)
 *  - 409 / 404 edge-cases are handled inside the abstraction
 *  - Token auth + base URL are wired once in the K8s layer, not per-call
 */

import { Effect, Layer } from "effect"
import { FileSystem } from "effect/FileSystem"
import * as Actor from "@de-workflow/xstate"
import {
  makeCreatePod,
  layer as K8sLayer,
  K8sHttpClient,
} from "effect/unstable/cluster/K8sHttpClient"
import { FetchHttpClient, HttpClient } from "effect/unstable/http"
import type * as v1 from "kubernetes-types/core/v1"
import { layerK8sHttpClient } from "@effect/platform-bun/BunClusterSocket"
import { BunFileSystem } from "@effect/platform-bun"

// ---------------------------------------------------------------------------
// Pod spec
// ---------------------------------------------------------------------------

const NAMESPACE = "default"
const POD_NAME  = "amazon-worker-import"
const IMAGE     = "workflowdeepecom.azurecr.io/import:V0.2"

const podSpec: v1.Pod = {
  apiVersion: "v1",
  kind: "Pod",
  metadata: {
    name: POD_NAME,
    namespace: NAMESPACE,
    labels: { app: "amazon-worker" },
  },
  spec: {
    restartPolicy: "Always",
    containers: [
      {
        name: "worker",
        image: IMAGE,
        imagePullPolicy: "Always",
        ports: [{ containerPort: 3000, name: "runner" }],
        env: [
          {
            name: "RUNNER_HOST",
            value: `${POD_NAME}.workers-headless.${NAMESPACE}.svc.cluster.local`,
          },
          { name: "DB_HOST",     value: "postgres" },
          { name: "DB_USER",     value: "effect" },
          { name: "DB_PASSWORD", value: "effect" },
          { name: "DB_DATABASE", value: "effect" },
          { name: "DB_PORT",     value: "5432" },
          { name: "OTLP_ENDPOINT", value: "http://20.164.51.212:4318/v1/traces" },
        ],
        resources: {
          requests: { cpu: "250m",  memory: "250Mi" },
          limits:   { cpu: "1000m", memory: "1024Mi" },
        },
      },
    ],
  },
}
export const AppLayer = K8sLayer.pipe(
  Layer.provide(FetchHttpClient.layer),
  Layer.provide(BunFileSystem.layer),
)
// ---------------------------------------------------------------------------
// Actor
// ---------------------------------------------------------------------------

export const makeImportActor = () =>
  Actor.fromEffect(() =>
    Effect.gen(function* () {
      const createPod = yield* makeCreatePod
      const status    = yield* createPod(podSpec)
      yield* Effect.log(`[${POD_NAME}] Pod is up — IP: ${status.podIP}`)
      return status
    }).pipe(
      Effect.scoped
    )
  )
  ````
