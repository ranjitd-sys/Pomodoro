```
import { BunHttpServer, BunRuntime } from "@effect/platform-bun"
import { Layer, pipe } from "effect"
import { RpcGroup, RpcServer, RpcSerialization } from "effect/unstable/rpc"
import { HttpRouter, HttpMiddleware } from "effect/unstable/http"
import { GetstreamContracts } from "./contract"
import { GetstreamLive } from "./handerl"

BunRuntime.runMain(
  pipe(
    HttpRouter.serve(
      RpcServer.layerHttp({
        path: "/token/rpc",
        protocol: "http",
        group: GetstreamContracts
      }),
      {
        middleware: HttpMiddleware.cors({
          allowedOrigins: ["*"],
          allowedMethods: ["POST", "GET", "OPTIONS"],
          allowedHeaders: ["Content-Type", "Authorization"],
        })
      }
    ),
    Layer.provide(GetstreamLive),
    Layer.provide(RpcSerialization.layerNdjson), 
    Layer.provide(BunHttpServer.layer({ hostname: "0.0.0.0", port: 3000 })),
    Layer.launch
  )
)
```
