1. [x] **Writting service a send email if user is  unauthorized**
```
import { Context, Effect, Layer } from "effect";
import nodemailer from "nodemailer";

export class EmailService extends Context.Service<EmailService>()(
  "email/EmailService",
  {
    make: Effect.gen(function* () {
      const host = yield* Effect.sync(() => process.env.SMTP_HOST).pipe(
        Effect.flatMap(Effect.fromNullishOr),
        Effect.mapError(() => new Error("Missing SMTP_HOST env var")),
      );

      const user = yield* Effect.sync(() => process.env.SMTP_USER).pipe(
        Effect.flatMap(Effect.fromNullishOr),
        Effect.mapError(() => new Error("Missing SMTP_USER env var")),
      );

      const pass = yield* Effect.sync(() => process.env.SMTP_PASS).pipe(
        Effect.flatMap(Effect.fromNullishOr),
        Effect.mapError(() => new Error("Missing SMTP_PASS env var")),
      );

      const transporter = nodemailer.createTransport({
        host,
        port: 587,
        auth: { user, pass },
      });

      yield* Effect.log("Email transporter created ✓");

      const send = (opts: {
        to: string;
        subject: string;
        body: string;
      }) =>
        Effect.tryPromise({
          try: () =>
            transporter.sendMail({
              from: user,
              to: opts.to,
              subject: opts.subject,
              html: opts.body,
            }),
          catch: (e) => new Error(`sendEmail failed: ${e}`),
        });

      return { send };
    }),
  },
) {
  static readonly layer = Layer.effect(EmailService, EmailService.make);
}
```
