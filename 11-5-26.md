2. [x] **Deploying on k8s and try to spawn on from Maruti**
```
import { Effect, Schedule } from "effect";
import { assign, createActor, createMachine } from "xstate";
import * as stream from "@de-workflow/getstream";
import * as Actor from "@de-workflow/xstate";
import { makeImportActor } from "../k8s/Import";
import * as k8s from "@kubernetes/client-node";

const NAMESPACE = "default";
const POD_NAME = "amazon-worker-import";
const IMAGE = "workflowdeepecom.azurecr.io/import:V0.2";

const Maruti = Effect.gen(function* () {
  const { client } = yield* stream.StreamService;

  let marutiActor: any;

  const watchActor = yield* Actor.fromEffect(() =>
    Effect.gen(function* () {
      yield* Effect.tryPromise({
        try: () =>
          client.connectUser(
            { id: "ranjit-dev" },
            client.createToken("ranjit-dev"),
          ),
        catch: (e) => new Error(`connectUser failed: ${e}`),
      });

      const channel = client.channel("messaging", "message");

      yield* Effect.tryPromise({
        try: () => channel.watch(),
        catch: (e) => new Error(`watch failed: ${e}`),
      });

      yield* Effect.log("Channel watching ✓");

      channel.on("message.new", (event) => {
        marutiActor.send({
          type: "MESSAGE_RECEIVED",
          message: event.message?.text,
        });
      });

      yield* Effect.never;
    }),
  );

  const processActor = yield* Actor.fromEffect(() =>
    Effect.gen(function* () {
      yield* Effect.log("processing...");
      yield* Effect.sleep("1 second");
    }),
  );

  const importActorLogic = yield* makeImportActor();
  
  const MarutiMachine = createMachine({
  id: "Maruti",
  type: "parallel",
  
  context: {
    amazonRef: null as any,
  },
  states: {

    watcher: {
      invoke: {
        src: watchActor,
        onError: {
          actions: ({ event }) => console.error("watchActor error:", event.error),
        },
      },
    },

  
    processor: {
      invoke: {
        src: processActor,
        onError: {
          actions: ({ event }) => console.error("processActor error:", event.error),
        },
      },
    },

    importFlow: {
      initial: "idle",
      states: {
        idle: {
          on: {
            MESSAGE_RECEIVED: {
              guard: ({ event }) => event.message === "Amazon",
              target: "importing",
              actions: () => console.log("Amazon message received, starting import..."),
            },
          },
        },
        importing: {
          invoke: {
            src: importActorLogic,
            onDone: {
              target: "idle",
              actions: () => console.log("Import done"),
            },
            onError: {
              target: "idle",
              actions: ({ event }) => console.error("Import error:", event.error),
            },
          },
        },
      },
    },
  },
});

  marutiActor = createActor(MarutiMachine, {
  
}).start();

}).pipe(Effect.provide(stream.StreamService.layer));
Effect.runPromise(Maruti);
```
