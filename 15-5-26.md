export const RunnerAddressConfig = RunnerAddress.RunnerAddress.makeUnsafe({ 
    host: process.env["RUNNER_HOST"] ?? "localhost", 
    port: 3000 
  })
export const ShardingConfigLive = ShardingConfig.layer({
  shardsPerGroup: 50,
  entityMailboxCapacity: 10,
  entityTerminationTimeout: 30_000,
  entityMessagePollInterval: 100,
  sendRetryInterval: 100,
  runnerAddress: Option.some(RunnerAddressConfig),
  shardGroups: ["amazon"],  // ← correct field name
})


export const WorkflowEngine = ClusterWorkflowEngine.layer.pipe(
  Layer.provideMerge(Sharding.layer),
  Layer.provideMerge(
    HttpRunner.layerHttp.pipe(
      Layer.provide(
        BunHttpServer.layerConfig({
          port: Config.withDefault(Config.int("RUNNER_PORT"), 3000),
        }),
      ),
      Layer.provide(BunHttpClient.layer),
      Layer.provide(RpcSerialization.layerNdjson),
    ),
  ),
  Layer.provideMerge(
    SqlMessageStorage.layer.pipe(
      Layer.provide(Snowflake.layerGenerator),
      Layer.provide(SqlLayer),
    ),
  ),
  Layer.provide(
    SqlRunnerStorage.layer.pipe(
      Layer.provide(Snowflake.layerGenerator),
      Layer.provide(SqlLayer),
    ),
  ),
  Layer.provide(RunnerHealth.layerNoop),
  Layer.provide(ShardingConfigLive),
)

const program = Effect.gen(function* () {
  yield* Effect.logInfo(" Aamazon Worker is booting up ");

  yield* Effect.sleep("500 millis");

  yield* Effect.logInfo("OTEL Wotking!");
  yield* Effect.never;
});

const MainLayer = ImportWorkflowlayer.pipe(
  Layer.provide(WorkflowEngine),
  Layer.provide(SqlLayer),
)

program.pipe(
  Effect.provide(MainLayer),
  BunRuntime.runMain,
)
