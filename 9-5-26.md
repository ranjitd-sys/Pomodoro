import { Config, Effect, Layer, Option } from "effect";

});

const TracingLive = NodeSdk.layer(
  Effect.sync(() => ({
    resource: {
      serviceName: "Workflow",
    },
    spanProcessor: new BatchSpanProcessor(traceExporter),
  })),
);

const ImportWorkflowlayer = Import.toLayer(

          yield* Effect.log(payload, executionId);
        }),
      });
    });
    return { clientId: Date.now.toString()};
  }),
);

const ShardingConfigLive = ShardingConfig.layer({
  shardsPerGroup: 50,
  entityMailboxCapacity: 10,
  entityTerminationTimeout: 30_000,
  entityMessagePollInterval: 1000,
  sendRetryInterval: 100,
  runnerAddress: Option.some( RunnerAddressConfig)
});


const WorkflowEngine = ClusterWorkflowEngine.layer.pipe(
  Layer.provideMerge(
    HttpRunner.layerHttp.pipe(
      Layer.provide(
        BunHttpServer.layerConfig({
          port: Config.withDefault(Config.int("RUNNER_PORT"), 2999),
        }),
      ),
    Layer.provide(BunHttpClient.layer),
    
      Layer.provide(RpcSerialization.layerNdjson),
    ),
  ),
  Layer.provideMerge(
    SqlMessageStorage.layer.pipe(
      Layer.provide(Snowflake.layerGenerator),
      Layer.provide(SqlLayer),
    ),
  ),
  Layer.provide(
    SqlRunnerStorage.layer.pipe(
      Layer.provide(Snowflake.layerGenerator),
      Layer.provide(SqlLayer),
    ),
  ),
  Layer.provide(RunnerHealth.layerNoop),
  Layer.provide(ShardingConfigLive),
);

const program = Effect.gen(function* () {
  yield* Effect.logInfo(" Aamazon Worker is booting up ");

  yield* Effect.sleep("500 millis");


  yield* Effect.logInfo("OTEL Wotking!");
  yield* Effect.never;
});

export const main_workflow = Layer.mergeAll(
  ImportWorkflowlayer,

)

program.pipe(
  Effect.provide(
    main_workflow.pipe(
      Layer.provideMerge(WorkflowEngine),
      Layer.provideMerge(SqlLayer),
    ),
  ),
  Effect.provide(TracingLive),
  BunRuntime.runMain,
);
