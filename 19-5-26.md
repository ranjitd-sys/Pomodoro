  const scaleupActor = yield* Actor.fromEffect(() =>
    Effect.gen(function* () {
      const count = yield* Ref.get(podCount);
      yield* Effect.log(`[ScaleUp] pod ${count + 1}/3 spawning...`);

      const res = yield* k8sRequest(`/api/v1/namespaces/${NAMESPACE}/pods`, {
        method: "POST",
        body: makePodSpec(count),
      });

      yield* Ref.update(podCount, (n) => n + 1);
      yield* Effect.log(`[ScaleUp] Done  total=${count + 1}`);
    }).pipe(
      Effect.provide(AppLayer),
      Effect.catchCause((e) =>
        Effect.log(`[ScaleUp] FATAL: ${JSON.stringify(e)}`),
      ),
    ),
  );

  const scaledownActor = yield* Actor.fromEffect(() =>
    Effect.gen(function* () {
      const count = yield* Ref.get(podCount);

      if (count === 0) {
        yield* Effect.log(`[ScaleDown] no pods to delete, skipping`);
        return;
      }

      const podIndex = count - 2;
      const podName = getPodName(podIndex);

      yield* Effect.log(`[ScaleDown] deleting pod ${podName}...`);

      yield* k8sRequest(`/api/v1/namespaces/${NAMESPACE}/pods/${podName}`, {
        method: "DELETE",
      }).pipe(
        Effect.catchCause((e) =>
          Effect.log(`[ScaleDown] delete failed: ${JSON.stringify(e)}`),
        ),
      );

      yield* Ref.update(podCount, (n) => Math.max(0, n - 1));
      yield* Effect.log(`[ScaleDown] Done  total=${podIndex}`);
    }).pipe(
      Effect.provide(AppLayer),
      Effect.catchCause((e) =>
        Effect.log(`[ScaleDown] FATAL: ${JSON.stringify(e)}`),
      ),
    ),
  );
