const k8sRequest = (
  path: string,
  options?: { method?: HttpMethod.HttpMethod; body?: unknown },
) =>
  Effect.gen(function* () {
    const client = yield* HttpClient.HttpClient;
    const token = getToken();
    const method: HttpMethod.HttpMethod = options?.method ?? "GET";
    const url = `${API_SERVER}${path}`;

    const baseReq = HttpClientRequest.make(method)(url);

    const req = options?.body
      ? yield* HttpClientRequest.bodyJson(options.body)(baseReq)
      : baseReq;

    const res = yield* client.execute(
      req.pipe(
        HttpClientRequest.setHeader("Authorization", `Bearer ${token}`),
        HttpClientRequest.setHeader("Content-Type", "application/json"),
      ),
    );

    return yield* res.json;
  });
