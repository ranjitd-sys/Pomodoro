import { Effect, Exit, MutableHashMap, Option } from "effect";
import * as Tracer from "effect/Tracer";
import {backgroundTelemetryLayer} from "../ext/src/background/backgroundMachineEffect"
export const tabSpans = MutableHashMap.empty<number, Tracer.Span>();

const addTrace = (
  tabId: number,
  spanName: "ui" | "backend" | "extension",
  eventName: string,
  attributes: Record<string, string> = {},
  rootSpan: Tracer.Span,
) =>
  Effect.gen(function* () {
    yield* Effect.log(`[${spanName}] ${eventName} for tab ${tabId}`).pipe(
      Effect.withSpan(eventName, {
        attributes: {
          ...attributes,
          "tab.id": String(tabId),
          "span.category": spanName,
        },
      }),
      Effect.withParentSpan(rootSpan),
    );
  });

export const traceUi = (
  tabId: number,
  eventName: string,
  rootSpan: Tracer.Span,
  attributes?: Record<string, string>,
) => addTrace(tabId, "ui", eventName, attributes, rootSpan);

export const traceBackend = (
  tabId: number,
  eventName: string,
  rootSpan: Tracer.Span,
  attributes?: Record<string, string>,
) => Effect.runPromise(
  addTrace(tabId, "backend", eventName, attributes, rootSpan)
) 

export const traceExtension = (
  tabId: number,
  eventName: string,
  rootSpan: Tracer.Span,
  attributes?: Record<string, string>,
) =>
  Effect.runPromise(
    addTrace(tabId, "extension", eventName, attributes, rootSpan).pipe(Effect.provide(backgroundTelemetryLayer))
  );

export const endTabSpan = (tabId: number) => {
  const existing = MutableHashMap.get(tabSpans, tabId);
  if (Option.isSome(existing)) {
    existing.value.end(BigInt(Date.now()) * 1_000_000n, Exit.void);
    MutableHashMap.remove(tabSpans, tabId);
  }
};
