import { Effect } from "effect";
import {  SenderLayer__ } from "../cluster/amazon_test/client";

import { BunRuntime } from "@effect/platform-bun";
import { Build, Export, Import, Write } from "./workflowDefination/amazonWorkflow";
import { WorkflowEngine } from "./amazonImport";
import { Build_Flipkart, Export_Flipkart, Import_Flipkart } from "./workflowDefination/flipakartWorkflow";
import { ClientTracingEnabled } from "effect/unstable/cluster/ClusterSchema";
import { Workflow } from "effect/unstable/workflow"
import { SqlLayer } from "../cluster/layers/sql";

const clientlist = new Map<number, any[]>();
clientlist.set(1, [Build, Import, Build_Flipkart])
clientlist.set(2, [Export, Import, Import_Flipkart , Build_Flipkart ])
clientlist.set(3, [Write, Export, Build_Flipkart , Export_Flipkart ])

const clients = Array.from(clientlist.entries());
const program = Effect.gen(function* () {
  yield* Effect.logInfo("Worker booting...");
  yield* Effect.sleep("500 millis");
    yield* Effect.forEach(clients, ([clientId, workflow])=>
    Effect.gen(function* (){
      for (const step of workflow) {
        const res = yield* step.execute({ clientId });
        console.log(`Client ${clientId} Step Result:`, res);
      }
      console.log(`================================== client ${clientId} Finished ==================================`);
    }),{concurrency:"unbounded"}
  )
  
});

program.pipe(
  Effect.provide(SenderLayer__),
  Effect.provide(SqlLayer),
  Effect.catchCause((e) => Effect.logError(e)),
  BunRuntime.runMain
)
