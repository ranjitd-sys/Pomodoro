import { Effect } from "effect";
import { SqlClient } from "effect/unstable/sql/SqlClient";

export const initDb = Effect.gen(function* () {
  const sql = yield* SqlClient;

  yield* sql`DROP TABLE IF EXISTS event_log`;

  yield* sql`
    CREATE TABLE IF NOT EXISTS event_log (
  id     BIGSERIAL PRIMARY KEY,
  stream_id   TEXT NOT NULL,
  seq_num     TEXT,
  event_name  TEXT,
  args        JSONB NOT NULL DEFAULT '{}',
  created_at  TIMESTAMPTZ DEFAULT now()
    )
  `;
  yield* sql`CREATE INDEX IF NOT EXISTS event_log_stream_idx ON event_log (stream_id, id)`;
  yield* sql`CREATE INDEX IF NOT EXISTS event_log_name_idx ON event_log (event_name)`;
});
