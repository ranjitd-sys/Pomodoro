#!/usr/bin/env python3
"""
PDF Invoice Processor for Kubernetes PVC
Watches /data/input/ for PDFs and processes them automatically
"""

import os
import time
import logging
from pathlib import Path
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

# Import the refactored PDF extractor
from firstcry_extractor import process_pdf, setup_logging


# ---------------------------------------------------------------------------
# Configuration
# ---------------------------------------------------------------------------

INPUT_DIR = "/data/input"
OUTPUT_DIR = "/data/output"
PROCESSED_DIR = "/data/processed"

# Create directories if they don't exist
for directory in [INPUT_DIR, OUTPUT_DIR, PROCESSED_DIR]:
    Path(directory).mkdir(parents=True, exist_ok=True)

# Setup logging
logger = setup_logging(log_file=os.path.join(OUTPUT_DIR, "processing.log"))


# ---------------------------------------------------------------------------
# File event handler
# ---------------------------------------------------------------------------

class PDFFileHandler(FileSystemEventHandler):
    """Watch for new PDF files in the input directory"""

    def on_created(self, event):
        """Handle file creation event"""
        if event.is_directory:
            return

        if event.src_path.endswith('.pdf'):
            logger.info(f"📁 New PDF detected: {event.src_path}")
            # Small delay to ensure file is fully written
            time.sleep(1)
            process_pdf(event.src_path, OUTPUT_DIR, PROCESSED_DIR)

    def on_modified(self, event):
        """Handle file modification event (optional)"""
        if event.is_directory:
            return

        if event.src_path.endswith('.pdf'):
            # Only process if file size hasn't changed in last second
            # (indicates upload is complete)
            if self._is_file_stable(event.src_path):
                logger.info(f"📝 PDF stable and ready for processing: {event.src_path}")
                # Optional: reprocess modified PDFs
                # process_pdf(event.src_path, OUTPUT_DIR, PROCESSED_DIR)

    @staticmethod
    def _is_file_stable(filepath: str, check_time: float = 1.0) -> bool:
        """Check if file size hasn't changed (indicates upload complete)"""
        try:
            size1 = os.path.getsize(filepath)
            time.sleep(check_time)
            size2 = os.path.getsize(filepath)
            return size1 == size2
        except Exception as e:
            logger.warning(f"Could not check file stability: {e}")
            return True


# ---------------------------------------------------------------------------
# Main processing functions
# ---------------------------------------------------------------------------

def process_existing_pdfs():
    """Process any PDFs already in the input directory"""
    input_path = Path(INPUT_DIR)
    pdf_files = sorted(input_path.glob("*.pdf"))

    if pdf_files:
        logger.info(f"🔍 Found {len(pdf_files)} existing PDF(s) to process")
        for pdf_file in pdf_files:
            logger.info(f"Processing existing: {pdf_file.name}")
            process_pdf(str(pdf_file), OUTPUT_DIR, PROCESSED_DIR)
    else:
        logger.info("✅ No existing PDFs found in input directory")


def watch_input_directory():
    """Watch the input directory for new PDFs"""
    logger.info("=" * 70)
    logger.info("PDF Invoice Processor Started")
    logger.info("=" * 70)
    logger.info(f"👀 Watching for PDFs in: {INPUT_DIR}")
    logger.info(f"📁 Output directory: {OUTPUT_DIR}")
    logger.info(f"📁 Processed directory: {PROCESSED_DIR}")
    logger.info("=" * 70)

    event_handler = PDFFileHandler()
    observer = Observer()
    observer.schedule(event_handler, INPUT_DIR, recursive=False)
    observer.start()

    try:
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        logger.info("⏹️  Stopping observer...")
        observer.stop()
    observer.join()


def health_check():
    """Simple health check to verify directories exist"""
    for directory in [INPUT_DIR, OUTPUT_DIR, PROCESSED_DIR]:
        if not os.path.isdir(directory):
            logger.error(f"❌ Directory not found: {directory}")
            return False
    logger.info("✅ All directories accessible")
    return True


# ---------------------------------------------------------------------------
# Entry point
# ---------------------------------------------------------------------------

def main():
    """Main function"""
    try:
        # Verify directories
        if not health_check():
            logger.error("❌ Directory check failed!")
            return 1

        # Process any existing PDFs
        process_existing_pdfs()

        # Start watching for new PDFs
        watch_input_directory()

        return 0

    except Exception as e:
        logger.error(f"❌ Fatal error: {e}")
        import traceback
        logger.error(traceback.format_exc())
        return 1


if __name__ == "__main__":
    import sys
    sys.exit(main())
